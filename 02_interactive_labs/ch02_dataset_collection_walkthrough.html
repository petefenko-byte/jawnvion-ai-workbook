# MinHash LSH deduplication with datasketch
from datasketch import MinHash, MinHashLSH
import re

def get_shingles(text: str, k: int = 5) -> set:
    """Generate k-shingles from text for MinHash."""
    text = re.sub(r'\s+', ' ', text.lower())
    return {text[i:i+k] for i in range(len(text) - k + 1)}

def build_minhash(text: str, num_perm: int = 128) -> MinHash:
    m = MinHash(num_perm=num_perm)
    for shingle in get_shingles(text):
        m.update(shingle.encode('utf8'))
    return m

# Build LSH index — threshold=0.8 → 80% similarity = near-dup

# Quality heuristic filters (C4-style)
import re
from typing import Optional

def passes_quality(text: str) -> bool:
    """
    Returns True if the document passes all quality heuristics.
    Based on C4 / Dolma quality filters.
    """
    words = text.split()

    # Minimum and maximum length
    if not (50 <= len(words) <= 100_000):
        return False

    # Ends with terminal punctuation
    if text.rstrip()[-1] not in '.!?':
        return False

    # At least 80% of lines must end with letter or punctuation
    lines = [l for l in text.splitlines() if l.strip()]
    good = sum(1 for l in lines if re.search(r'[\w.!?]$', l))
    if good / max(len(lines), 1) < 0.8:
        return False

    # Curly-brace check — likely code or template noise
    if text.count('{') > 10:
        return False

    # "Lorem ipsum" lorem-flag
    if 'lorem ipsum' in text.lower():
        return False

    return True
