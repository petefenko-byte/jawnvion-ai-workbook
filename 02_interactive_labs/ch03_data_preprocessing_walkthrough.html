# Sliding window chunking for long documents
from typing import List, Dict
from transformers import AutoTokenizer

tokenizer = AutoTokenizer.from_pretrained("meta-llama/Meta-Llama-3-8B")

def chunk_document(
    text: str,
    chunk_size: int = 512,
    stride: int = 64,
) -> List[Dict]:
    """
    Tokenize then split into overlapping windows.
    Returns list of {input_ids, attention_mask, chunk_idx}.
    """
    tokens = tokenizer(text, return_tensors="pt")
    ids = tokens["input_ids"][0]
    total = len(ids)
    chunks = []
    idx = 0
    start = 0

    while start < total:
        end = min(start + chunk_size, total)
        chunk_ids = ids[start:end]
        chunks.append({
            "input_ids": chunk_ids,
            "chunk_idx": idx,
            "start_token": start,
            "end_token": end,
        })
        idx += 1
        start += chunk_size - stride   # slide by (chunk_size - overlap)
        if end == total:
            break

    print(f"Document: {total} tokens → {len(chunks)} chunks")
    return chunks
# Stratified train / val / test split
from sklearn.model_selection import train_test_split
import numpy as np

def split_dataset(
    dataset,
    train_pct: float = 0.8,
    val_pct:   float = 0.1,
    seed:      int   = 42,
    stratify_col: str = "label",
):
    """Stratified split preserving class distribution."""
    labels = dataset[stratify_col]

    # First split: train vs (val + test)
    train, temp = train_test_split(
        dataset,
        test_size=1 - train_pct,
        random_state=seed,
        stratify=labels,
    )

    # Second split: val vs test (from temp)
    val_ratio = val_pct / (1 - train_pct)
    val, test = train_test_split(
        temp,
        test_size=1 - val_ratio,
        random_state=seed,
        stratify=temp[stratify_col],
    )

    print(f"Train: {len(train)} | Val: {len(val)} | Test: {len(test)}")
    return train, val, test

# Text normalization pipeline
import re, unicodedata, html

def normalize_text(text: str) -> str:
    """
    Full normalization pipeline: unicode → html → whitespace → punct.
    Apply before tokenization.
    """
    # 1. Unicode normalization (NFC: canonical decomposition + composition)
    text = unicodedata.normalize('NFC', text)

    # 2. Strip HTML entities and tags
    text = html.unescape(text)
    text = re.sub(r'<[^>]+>', ' ', text)

    # 3. Remove control characters (except tab and newline)
    text = ''.join(c for c in text
                  if unicodedata.category(c)[0] != 'C'
                  or c in ('\t', '\n'))

    # 4. Collapse whitespace (preserve single newlines)
    text = re.sub(r'[^\S\n]+', ' ', text)   # spaces → single space
    text = re.sub(r'\n{3,}', '\n\n', text)  # 3+ newlines → 2

    # 5. Strip leading/trailing whitespace per line
    lines = [l.strip() for l in text.splitlines()]
    text = '\n'.join(l for l in lines if l)

    return text.strip()
