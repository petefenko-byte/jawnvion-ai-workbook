# Hardware availability check — run this first
import torch
import subprocess

print(f"PyTorch: {torch.__version__}")
print(f"CUDA available: {torch.cuda.is_available()}")

if torch.cuda.is_available():
    gpu = torch.cuda.get_device_properties(0)
    vram_gb = gpu.total_memory / 1e9
    print(f"GPU: {gpu.name}")
    print(f"VRAM: {vram_gb:.1f} GB")

    # Model size guidance
    if vram_gb >= 80:
        print("→ Can train up to 70B full precision or 405B quantized")
    elif vram_gb >= 24:
        print("→ Full fine-tune up to 7B; QLoRA up to 70B")
    elif vram_gb >= 12:
        print("→ QLoRA fine-tune up to 13B; full fine-tune up to 1.5B")
    else:
        print("→ API or GGUF inference recommended")
else:
    print("No GPU found — use API or CPU inference (GGUF/Ollama)")
# Establish a naive baseline before any model training
# This is your floor — the model must beat this to have value
from sklearn.metrics import f1_score, accuracy_score
import numpy as np

# Example: classification baseline (majority class predictor)
def majority_class_baseline(y_true):
    majority = np.bincount(y_true).argmax()
    y_pred = np.full_like(y_true, majority)
    acc = accuracy_score(y_true, y_pred)
    f1  = f1_score(y_true, y_pred, average='macro')
    print(f"Majority class: {majority}")
    print(f"Baseline Accuracy: {acc:.4f}")
    print(f"Baseline Macro-F1: {f1:.4f}")
    print(f"Your model must beat F1={f1:.2f} to add value.")
    return {'accuracy': acc, 'f1': f1}

# Example: generation baseline (perplexity of base model, no fine-tuning)
from transformers import AutoModelForCausalLM, AutoTokenizer
import math, torch

def compute_baseline_ppl(texts, model_name="gpt2"):
    tokenizer = AutoTokenizer.from_pretrained(model_name)
    model = AutoModelForCausalLM.from_pretrained(model_name).eval()
    nlls = []
    for text in texts[:50]:  # sample 50 docs
        inputs = tokenizer(text, return_tensors="pt", max_length=512, truncation=True)
        with torch.no_grad():
            loss = model(**inputs, labels=inputs.input_ids).loss
        nlls.append(loss.item())
    ppl = math.exp(np.mean(nlls))
    print(f"Baseline Perplexity ({model_name}): {ppl:.2f}")
    print("Fine-tuned model should achieve lower perplexity on your domain.")
    return ppl
# Metric setup for all task types — run once at project start
# pip install evaluate sacrebleu rouge_score bert-score
import evaluate

TASK_METRICS = {
    'classification': ['accuracy', 'f1'],
    'ner':            ['seqeval'],
    'qa':             ['squad'],
    'summarization':  ['rouge', 'bertscore'],
    'translation':    ['sacrebleu'],
    'generation':     ['perplexity'],
}

def load_metrics(task: str):
    if task not in TASK_METRICS:
        raise ValueError(f"Unknown task: {task}. Choose from {list(TASK_METRICS)}")
    metrics = {}
    for name in TASK_METRICS[task]:
        metrics[name] = evaluate.load(name)
        print(f"✓ Loaded metric: {name}")
    return metrics

# Usage:
metrics = load_metrics('summarization')
# → ✓ Loaded metric: rouge
# → ✓ Loaded metric: bertscore
